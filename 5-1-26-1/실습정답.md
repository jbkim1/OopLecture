# 종합 실습문제 - 정답 (클래스 선언/정의 + 상속 + 가상 함수)

---

**Payment.h**
```cpp
#pragma once
#include <string>

class Payment {
private:
    int amount;

public:
    Payment(int amount);
    int getAmount() const;
    virtual void printReceipt() const;
};
```

**Payment.cpp**
```cpp
#include "Payment.h"
#include <iostream>

Payment::Payment(int amount) : amount(amount) {}

int Payment::getAmount() const {
    return amount;
}

void Payment::printReceipt() const {
    std::cout << amount << "원" << std::endl;
}
```

**CardPayment.h**
```cpp
#pragma once
#include "Payment.h"
#include <string>

class CardPayment : public Payment {
private:
    std::string cardNumber;

public:
    CardPayment(int amount, const std::string& cardNumber);
    std::string getCardNumber() const;
    void process() const;
    void printReceipt() const override;
};
```

**CardPayment.cpp**
```cpp
#include "CardPayment.h"
#include <iostream>

CardPayment::CardPayment(int amount, const std::string& cardNumber)
    : Payment(amount), cardNumber(cardNumber) {}

std::string CardPayment::getCardNumber() const {
    return cardNumber;
}

void CardPayment::process() const {
    std::cout << "카드 결제 처리 중... 완료" << std::endl;
}

void CardPayment::printReceipt() const {
    std::cout << "[카드] " << getAmount() << "원" << std::endl;
    process();
    std::cout << "카드번호: " << cardNumber << std::endl;
}
```

**CashPayment.h**
```cpp
#pragma once
#include "Payment.h"

class CashPayment : public Payment {
private:
    int receivedAmount;

public:
    CashPayment(int amount, int receivedAmount);
    int getReceivedAmount() const;
    void process() const;
    void printReceipt() const override;
};
```

**CashPayment.cpp**
```cpp
#include "CashPayment.h"
#include <iostream>

CashPayment::CashPayment(int amount, int receivedAmount)
    : Payment(amount), receivedAmount(receivedAmount) {}

int CashPayment::getReceivedAmount() const {
    return receivedAmount;
}

void CashPayment::process() const {
    std::cout << "현금 결제 처리 중... 완료" << std::endl;
}

void CashPayment::printReceipt() const {
    std::cout << "[현금] " << getAmount() << "원" << std::endl;
    process();
    std::cout << "거스름돈: " << (receivedAmount - getAmount()) << "원" << std::endl;
}
```

**main.cpp**
```cpp
#include "CardPayment.h"
#include "CashPayment.h"
#include <iostream>

int main() {
    CardPayment card(15000, "1234-5678-9012-3456");
    CashPayment cash(8000, 10000);

    Payment* payments[2];
    payments[0] = &card;
    payments[1] = &cash;

    for (int i = 0; i < 2; i++) {
        payments[i]->printReceipt();
        std::cout << "---" << std::endl;
    }

    return 0;
}
```
